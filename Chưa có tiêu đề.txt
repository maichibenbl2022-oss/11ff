-- Hệ thống mã hóa tín hiệu ẩn (Zero-Width)
local OBS_SIGNATURE = "HMOBS"
local ZW_ZERO = "\u{200B}"
local ZW_ONE = "\u{200C}"

local function EncodeSignal(Command, UserId, Extra)
    local SignalString = OBS_SIGNATURE .. "|" .. tostring(Command) .. "|" .. tostring(UserId)
    if Extra then
        local SanitizedExtra = tostring(Extra):gsub("|", "~")
        SignalString = SignalString .. "|" .. SanitizedExtra
    end
    local OutputTable = {}
    for Index = 1, #SignalString do
        local ByteChar = string.byte(SignalString, Index)
        for BitIndex = 7, 0, -1 do
            local BitValue = math.floor(ByteChar / 2 ^ BitIndex) % 2
            if BitValue == 1 then
                table.insert(OutputTable, ZW_ONE)
            else
                table.insert(OutputTable, ZW_ZERO)
            end
        end
    end
    return table.concat(OutputTable)
end

local function DecodeSignal(Text)
    if not Text then return nil end
    if #Text < 64 then return nil end
    local BitsTable = {}
    for Index = 1, #Text do
        local Char = string.sub(Text, Index, Index)
        if Char == ZW_ZERO then
            table.insert(BitsTable, "0")
        elseif Char == ZW_ONE then
            table.insert(BitsTable, "1")
        end
    end
    if #BitsTable < 8 then return nil end
    local OutputString = {}
    for Index = 1, #BitsTable - 7, 8 do
        local ByteValue = 0
        for BitIndex = 0, 7 do
            local BitString = BitsTable[Index + BitIndex]
            ByteValue = ByteValue * 2 + tonumber(BitString)
        end
        table.insert(OutputString, string.char(ByteValue))
    end
    local DecodedString = table.concat(OutputString)
    if string.sub(DecodedString, 1, #OBS_SIGNATURE) ~= OBS_SIGNATURE then return nil end
    local SplitParts = string.split(DecodedString, "|")
    if #SplitParts < 3 then return nil end
    local ExtraData = SplitParts[4]
    if ExtraData then
        ExtraData = string.gsub(ExtraData, "~", "|")
    end
    return SplitParts[2], tonumber(SplitParts[3]), ExtraData
end

local function ValidPlayer(Player)
    if not Player or Player == LocalPlayer or Allies[Player.Name] then return false end
    local Char = Player.Character
    if not Char then return false end
    local Root = Char:FindFirstChild("HumanoidRootPart")
    local Hum = Char:FindFirstChildOfClass("Humanoid")
    return Root and Hum and Hum.Health > 0
end

local function FindNearest(Range)
    local MyRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not MyRoot then return nil end
    local Best, BestDist = nil, Range or 100
    for _, Player in ipairs(PlayersService:GetPlayers()) do
        if ValidPlayer(Player) then
            local Root = Player.Character:FindFirstChild("HumanoidRootPart")
            if Root then
                local Dist = (MyRoot.Position - Root.Position).Magnitude
                if Dist < BestDist then
                    BestDist = Dist
                    Best = Player
                end
            end
        end
    end
    return Best
end

local function FindByName(Name)
    if not Name or Name == "" then return nil end
    local LowerName = string.lower(Name)
    for _, Player in ipairs(PlayersService:GetPlayers()) do
        if Player ~= LocalPlayer and string.find(string.lower(Player.Name), LowerName, 1, true) then
            return Player
        end
    end
    return nil
end

local function GetTool()
    local Char = LocalPlayer.Character
    if not Char then return nil end
    local Tool = Char:FindFirstChildOfClass("Tool")
    if Tool then return Tool end
    local Backpack = LocalPlayer:FindFirstChild("Backpack")
    if Backpack then
        for _, Item in ipairs(Backpack:GetChildren()) do
            if Item:IsA("Tool") then
                Item.Parent = Char
                return Item
            end
        end
    end
    return nil
end

local function HitAll(Target)
    local Char = LocalPlayer.Character
    if not Char then return end
    for _, Tool in ipairs(Char:GetChildren()) do
        if Tool:IsA("Tool") then
            pcall(function()
                Tool.Enabled = true
                Tool:Activate()
            end)
        end
    end
    for _, Remote in ipairs(Remotes) do
        pcall(function() Remote:FireServer("M1") end)
        pcall(function() Remote:FireServer() end)

    for _, Remote in ipairs(ManualRemotes) do
        pcall(function() Remote:FireServer("M1") end)
        pcall(function() Remote:FireServer("Attack") end)
    end
    if ClientSideHitbox and Target then
        pcall(function() ClientSideHitbox:FireServer(Target, buffer.fromstring("10069.74")) end)
    end
    if CombatRemote then
        pcall(function() CombatRemote:FireServer("M1") end)
        pcall(function() CombatRemote:FireServer("Attack") end)
    end
end

local function FindPlayerAdmin(Input)
    if not Input or Input == "" then return nil end
    local UID = tonumber(Input)
    if UID then
        local Target = PlayersService:GetPlayerByUserId(UID)
        if Target then return Target end
    end
    local LowerInput = string.lower(Input)
    for _, Player in ipairs(PlayersService:GetPlayers()) do
        if Player ~= LocalPlayer and string.find(string.lower(Player.Name), LowerInput, 1, true) then return Player end
    end
    return nil
end

local function IsMonster(Model)
    if not Model or not Model:IsA("Model") then return false end
    local Hum = Model:FindFirstChildOfClass("Humanoid")
    if not Hum or Hum.Health <= 0 then return false end
    if Model == LocalPlayer.Character then return false end
    if PlayersService:GetPlayerFromCharacter(Model) then return false end
    return (Model:FindFirstChild("HumanoidRootPart") or Model.PrimaryPart) ~= nil
end

local function GetMonsterRoot(Model)
    return Model:FindFirstChild("HumanoidRootPart") or Model.PrimaryPart
end

local function ScanMonsters()
    local List = {}
    for _, V in ipairs(workspace:GetDescendants()) do
        if V:IsA("Model") and IsMonster(V) then table.insert(List, V) end
    end
    return List
end

local function FindTargetByNamePartial(Partial)
    if not Partial or Partial == "" then return nil end
    local Lower = string.lower(Partial)
    for _, Player in ipairs(PlayersService:GetPlayers()) do
        if Player ~= LocalPlayer and string.find(string.lower(Player.Name), Lower, 1, true) then return Player end
    end
    return nil
end

local function ClaimOwnership(Model)
    pcall(function()
        for _, Part in ipairs(Model:GetDescendants()) do
            if Part:IsA("BasePart") and Part.SetNetworkOwner then Part:SetNetworkOwner(LocalPlayer) end
        end
    end)
end

local function CollectAllMonsters()
    local MyChar = LocalPlayer.Character
    local MyRoot = MyChar and MyChar:FindFirstChild("HumanoidRootPart")
    if not MyRoot then return end
    local Found = ScanMonsters()
    local Count = 0
    for _, Model in ipairs(Found) do
        if Count >= Config.MonsterMax then break end
        local Root = GetMonsterRoot(Model)
        if Root then
            ClaimOwnership(Model)
            local Angle = (Count / math.max(1, #Found)) * math.pi * 2
            local Pos = MyRoot.Position + Vector3.new(
                math.cos(Angle) * Config.MonsterOrbitRadius,
                Config.MonsterOrbitHeight,
                math.sin(Angle) * Config.MonsterOrbitRadius
            )
            pcall(function()
                Model:PivotTo(CFrame.new(Pos))
                local Hum = Model:FindFirstChildOfClass("Humanoid")
                if Hum then Hum.WalkSpeed = 60 end
            end)
            Monsters[Model] = true
            Count = Count + 1
        end
    end
end

local function SummonAgainstTarget()
    local Target = Config.MonsterTarget
    if not Target or not Target.Character then return end
    local TRoot = Target.Character:FindFirstChild("HumanoidRootPart")
    if not TRoot then return end
    local Found = ScanMonsters()
    local Count = 0
    for _, Model in ipairs(Found) do
        if Count >= Config.MonsterMax then break end
        local Root = GetMonsterRoot(Model)
        local Hum = Model:FindFirstChildOfClass("Humanoid")
        if Root and Hum then
            ClaimOwnership(Model)
            if Config.MonsterMode == "TP" then
                local Offset = Vector3.new(math.random(-3, 3), 2, math.random(-3, 3))
                pcall(function() Model:PivotTo(CFrame.new(TRoot.Position + Offset)) end)
            end
            Monsters[Model] = true
            Count = Count + 1
        end
    end
end
